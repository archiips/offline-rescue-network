cmake_minimum_required(VERSION 3.20)
project(RescueWorkflowModel LANGUAGES CXX)
option(RESCUE_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_library(rescue_workflow src/workflow.cpp)
target_include_directories(rescue_workflow PUBLIC include)
target_compile_features(rescue_workflow PUBLIC cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(rescue_workflow PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(RESCUE_SANITIZERS)
    target_compile_options(rescue_workflow PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(rescue_workflow PUBLIC -fsanitize=address,undefined)
  endif()
elseif(RESCUE_SANITIZERS)
  message(FATAL_ERROR "RESCUE_SANITIZERS requires Clang or GNU")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(workflow_tests tests/workflow_tests.cpp)
  target_link_libraries(workflow_tests PRIVATE rescue_workflow)
  foreach(scenario roundtrip updates handling ownership references failures ordering limits late_gap repeated_withdrawal dependency_retry invalid_handling hidden_update)
    add_test(NAME workflow_${scenario} COMMAND workflow_tests ${scenario})
  endforeach()
endif()
add_executable(workflow_demo src/demo.cpp)
target_link_libraries(workflow_demo PRIVATE rescue_workflow)

find_package(SQLite3 REQUIRED)
add_library(rescue_demo_bridge bridge/workflow_bridge.cpp bridge/session_store.cpp bridge/endpoint.cpp bridge/event_wire.cpp)
target_include_directories(rescue_demo_bridge PUBLIC bridge/include)
if(NOT TARGET SQLite3::SQLite3)
  add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()
target_link_libraries(rescue_demo_bridge PUBLIC rescue_workflow SQLite3::SQLite3)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(rescue_demo_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(bridge_tests tests/bridge_tests.cpp)
  target_link_libraries(bridge_tests PRIVATE rescue_demo_bridge)
  add_test(NAME bridge_boundary_queue_lifetime COMMAND bridge_tests)
endif()
if(BUILD_TESTING AND UNIX)
  add_executable(persistence_tests tests/persistence_tests.cpp)
  target_link_libraries(persistence_tests PRIVATE rescue_demo_bridge)
  foreach(scenario restart failure lock stale corrupt interrupted full commit_lock initialization wal)
    add_test(NAME persistence_${scenario} COMMAND persistence_tests ${scenario})
  endforeach()
endif()

if(BUILD_TESTING)
  add_executable(endpoint_tests tests/endpoint_tests.cpp)
  target_link_libraries(endpoint_tests PRIVATE rescue_demo_bridge)
  target_include_directories(endpoint_tests PRIVATE bridge)
  foreach(scenario exchange retry locks isolation malformed capacity)
    add_test(NAME endpoint_${scenario} COMMAND endpoint_tests ${scenario})
  endforeach()
endif()

add_library(rescue_relay bridge/relay_queue.cpp)
target_include_directories(rescue_relay PUBLIC bridge/include)
target_compile_features(rescue_relay PUBLIC cxx_std_20)
target_link_libraries(rescue_relay PUBLIC SQLite3::SQLite3)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(rescue_relay PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(RESCUE_SANITIZERS)
    target_compile_options(rescue_relay PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(rescue_relay PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(BUILD_TESTING AND UNIX)
  add_executable(relay_tests tests/relay_tests.cpp)
  target_link_libraries(relay_tests PRIVATE rescue_relay rescue_demo_bridge)
  foreach(scenario durable duplicate fifo fairness capacity bounds locks tamper incompatible descriptors journal lookup cache)
    add_test(NAME relay_${scenario} COMMAND relay_tests ${scenario})
  endforeach()
endif()
